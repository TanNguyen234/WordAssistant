import React, { useState, useEffect } from 'react';
import {
  FileText,
  CheckSquare,
} from 'lucide-react';
import { useAuth } from '../auth/AuthContext';
import { api } from '../api/client';
import { formatScore } from '../utils/vi';
import type { Grade, Course } from '../types';

export const GradesPage: React.FC = () => {
  const { user } = useAuth();
  const [courses, setCourses] = useState<Course[]>([]);
  const [selectedCourseId, setSelectedCourseId] = useState<string>('');
  const [grades, setGrades] = useState<Grade[]>([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    const init = async () => {
      try {
        setLoading(true);
        const courseList = await api.getCourses();
        setCourses(courseList);
        if (courseList.length > 0) {
          setSelectedCourseId(String(courseList[0].id));
        }
      } catch (err) {
        console.error(err);
      } finally {
        setLoading(false);
      }
    };
    init();
  }, [user]);

  const fetchGrades = async () => {
    if (!selectedCourseId) return;
    try {
      setLoading(true);
      if (user?.role === 'STUDENT') {
        const myGrades = await api.getMyGrades(selectedCourseId);
        setGrades(myGrades);
      } else {
        const book = await api.getCourseGradebook(selectedCourseId);
        setGrades(book);
      }
    } catch (err) {
      console.error(err);
    } finally {
      setLoading(false);
    }
  };

  useEffect(() => {
    if (selectedCourseId) {
      fetchGrades();
    }
  }, [selectedCourseId, user]);

  const calculateStudentAverage = () => {
    if (grades.length === 0) return 0;
    const total = grades.reduce((acc, g) => acc + (g.score / g.max_score) * 10, 0);
    return total / grades.length;
  };

  const getSourceBadge = (type: string) => {
    switch (type) {
      case 'ASSIGNMENT':
        return (
          <span className="inline-flex items-center text-xs font-semibold px-2 py-0.5 rounded-md bg-blue-100 text-blue-700">
            <FileText className="w-3 h-3 mr-1" /> Bài tập
          </span>
        );
      case 'QUIZ':
        return (
          <span className="inline-flex items-center text-xs font-semibold px-2 py-0.5 rounded-md bg-purple-100 text-purple-700">
            <CheckSquare className="w-3 h-3 mr-1" /> Bài kiểm tra
          </span>
        );
      default:
        return (
          <span className="inline-flex items-center text-xs font-semibold px-2 py-0.5 rounded-md bg-slate-100 text-slate-700">
            {type}
          </span>
        );
    }
  };

  return (
    <div className="space-y-6">
      {/* Header */}
      <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-4">
        <div>
          <h1 className="text-2xl font-bold text-slate-800 tracking-tight">
            {user?.role === 'STUDENT' ? 'Bảng điểm và kết quả học tập cá nhân' : 'Sổ điểm lớp học phần'}
          </h1>
          <p className="text-sm text-slate-500">
            {user?.role === 'STUDENT'
              ? 'Theo dõi kết quả đánh giá các cột điểm thành phần và nhận xét từ giảng viên'
              : 'Xem tổng hợp điểm số các cột điểm bài tập, bài kiểm tra và đánh giá quá trình'}
          </p>
        </div>
      </div>

      {/* Student KPI Card */}
      {user?.role === 'STUDENT' && (
        <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">
          <div className="bg-white p-5 rounded-2xl border border-slate-200/80 shadow-xs">
            <span className="text-xs font-semibold text-slate-400 uppercase">Điểm trung bình tạm tính</span>
            <div className="flex items-baseline space-x-2 mt-2">
              <span className="text-3xl font-extrabold text-blue-600">
                {formatScore(grades.length ? calculateStudentAverage() : null, 2)}
              </span>
              <span className="text-xs text-slate-400">/ 10</span>
            </div>
          </div>

          <div className="bg-white p-5 rounded-2xl border border-slate-200/80 shadow-xs">
            <span className="text-xs font-semibold text-slate-400 uppercase">Số cột điểm đã tích lũy</span>
            <div className="flex items-baseline space-x-2 mt-2">
              <span className="text-3xl font-extrabold text-slate-800">{grades.length}</span>
              <span className="text-xs text-slate-400">cột điểm</span>
            </div>
          </div>

          <div className="bg-white p-5 rounded-2xl border border-slate-200/80 shadow-xs">
            <span className="text-xs font-semibold text-slate-400 uppercase">Tình trạng ghi nhận điểm</span>
            <div className="flex items-baseline space-x-2 mt-2">
              <span className="text-xl font-bold text-emerald-600">
                {grades.length ? 'Đã có điểm thành phần' : 'Chưa có điểm thành phần'}
              </span>
            </div>
          </div>
        </div>

      {user?.role === 'STUDENT' && (
        <p className="text-sm text-slate-600">Điểm trung bình tạm tính từ các điểm đã ghi nhận, chưa áp dụng trọng số học phần. Kết quả này chưa xác định việc đạt học phần.</p>
      )}

      {/* Course Selector */}
      <div className="bg-white p-4 rounded-2xl border border-slate-200/80 shadow-xs flex items-center space-x-3">

          Khóa học:
        </label>
        <select

          onChange={(e) => setSelectedCourseId(e.target.value)}
          className="w-full sm:w-96 px-3.5 py-2 rounded-xl border border-slate-200 text-sm bg-white focus:outline-none focus:ring-2 focus:ring-blue-500/20 focus:border-blue-600 font-medium text-slate-800"
        >
          {courses.map((c) => (
            <option key={c.id} value={c.id}>
              {c.code} - {c.title}
            </option>
          ))}
        </select>
      </div>

      {/* Grades Table */}
      <div className="bg-white rounded-2xl border border-slate-200/80 shadow-xs overflow-hidden">
        <div className="overflow-x-auto">
          <table className="w-full text-left border-collapse text-sm">
            <thead>

                        </div>
                      ) : (
                        <div>
                          <div className="font-semibold text-slate-800">{g.student_name || 'Sinh viên'}</div>

                        </div>
                      )}
                    </td>
                    <td className="py-3.5 px-4">{getSourceBadge(g.sourceType || g.source_type || 'ASSIGNMENT')}</td>
                    <td className="py-3.5 px-4">
                      <span className="text-base font-extrabold text-blue-600">{g.score}</span>
                      <span className="text-xs text-slate-400"> / {g.max_score}</span>

                    </td>
                  </tr>
                ))
              )}
            </tbody>
          </table>
        </div>
      </div>
    </div>
  );
};

export function apiErrorMessageVi(status: number, detail = ''): string {
  if (status >= 500) return 'Máy chủ chưa xử lý được yêu cầu. Vui lòng thử lại sau.';
  if (Object.hasOwn(ERROR_LABELS, detail)) return ERROR_LABELS[detail];
  if (/Access denied|not authorized|do not own|Requires role/.test(detail)) return 'Bạn không có quyền thực hiện thao tác này.';
  // Already-localized server explanations stay intact; never translate saved user content.
  if (/[À-ỹĐđ]/u.test(detail)) return detail;
  switch (status) {
    case 0: return 'Không thể kết nối đến máy chủ. Vui lòng kiểm tra kết nối và thử lại.';
    case 401: return 'Phiên đăng nhập không hợp lệ. Vui lòng đăng nhập lại.';
    case 403: return 'Bạn không có quyền thực hiện thao tác này.';
